// Primer funkcije unutar index.html koja šalje zahtev na naš API async function generateScript() { const promptText = document.getElementById('promptInput').value; const errorDiv = document.getElementById('errorDiv'); const outputDiv = document.getElementById('outputDiv'); errorDiv.innerText = ""; outputDiv.innerText = "Generisanje u toku..."; try { const response = await fetch('/api/generate', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ prompt: promptText }) }); const data = await response.json(); if (!response.ok) { throw new Error(data.error || 'Došlo je do greške na serveru.'); } // Prikazujemo dobijeni tekst na ekranu outputDiv.innerText = data.output; } catch (err) { errorDiv.innerText = "Greška: " + err.message; outputDiv.innerText = ""; } }
