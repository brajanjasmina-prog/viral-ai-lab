async function handleGenerate() { const platform = document.getElementById('platform').value; // ili odakle već čitaš formu const niche = document.getElementById('niche').value; const promptText = document.getElementById('prompt').value; const fullPrompt = `Platform: ${platform}. Niche: ${niche}. Instructions: ${promptText}`; try { const response = await fetch('/api/generate', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ prompt: fullPrompt }) }); const data = await response.json(); if (!response.ok) { throw new Error(data.error || 'Greška na serveru'); } // Prikaz rezultata na ekranu console.log(data.output); alert(data.output); // Privremeno da vidiš rezultat } catch (error) { console.error('Greška:', error); alert('Greška: ' + error.message); } }
